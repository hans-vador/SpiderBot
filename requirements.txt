# Python dependencies for the Raspberry Pi side of SpiderBot (everything under RPI/).
#
#   pip install -r requirements.txt
#
# Create the venv with --system-site-packages (see docs/SETUP.md) so the GPIO
# libraries that Raspberry Pi OS ships via apt stay visible inside the venv.
# The Servo 2040 firmware under Servo2040/ needs nothing from this file.

RPi.GPIO>=0.7.1          # pincer servo PWM on GPIO 17             (stateMachine.py, tools/Pincer.py)
gpiozero>=2.0            # limit-switch input                       (tools/GPIOTest.py, legacy/Controller.py)
